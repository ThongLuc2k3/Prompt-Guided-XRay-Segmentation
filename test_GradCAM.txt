"""
test_inference_gradcam.py
=========================
Cải tiến inference:
  1. Phát hiện prompt sai (confidence thấp HOẶC tâm pred lệch xa tâm prompt)
  2. Khi nghi ngờ → kích hoạt GradCAM từ bottleneck để gợi ý vùng u thật sự
  3. Đo độ chính xác của gợi ý bằng chỉ số CBL (Khoảng cách Tâm u vs Điểm gợi ý)

Visualize: 2 rows (Prompt ĐÚNG | Prompt SAI) × 5 cols
  Col: Ảnh gốc | Prompt | Ground Truth | Dự đoán | GradCAM / Prob map

Cách chạy:
    python test_inference_gradcam.py
"""

import os
import cv2
import json
import torch
import torch.nn.functional as F
import numpy as np
import matplotlib.pyplot as plt
from scipy.ndimage import binary_erosion, distance_transform_edt

from dataset import BTXRD_Dataset
from models.networks.prompt_unet_2D import PGA_UNet

# =========================================================
# CẤU HÌNH – chỉnh tại đây
# =========================================================
DEVICE             = torch.device("cuda" if torch.cuda.is_available() else "cpu")
MODEL_PATH         = "checkpoints/pga_unet_expB_best.pth"
IMG_SIZE           = 512
USE_ENCODER_PROMPT = True

# Ngưỡng phát hiện prompt sai
CONFIDENCE_THRESHOLD = 0.30   # max(sigmoid(pred)) < ngưỡng → nghi ngờ
CENTER_DIST_RATIO    = 0.25   # dist(tâm_pred, tâm_prompt) > ratio*IMG_SIZE → nghi ngờ

TEST_IMAGE_DIR = "dataset_BTXRD/test/images"
TEST_JSON_DIR  = "dataset_BTXRD/test/annotations"

# Danh sách ảnh muốn chạy visualize
VIS_IMAGES = [
    "IMG001768.png",
    "IMG001538.png",
    "IMG001100.png",
]

# =========================================================
# METRIC HELPERS
# =========================================================

def calc_hd95(pred: np.ndarray, gt: np.ndarray) -> float:
    pred, gt = pred.astype(bool), gt.astype(bool)
    if not pred.any() and not gt.any():
        return 0.0
    if not pred.any() or not gt.any():
        return float(IMG_SIZE)
    pe = pred ^ binary_erosion(pred)
    ge = gt   ^ binary_erosion(gt)
    d1 = distance_transform_edt(~ge)[pe]
    d2 = distance_transform_edt(~pe)[ge]
    if not len(d1) or not len(d2):
        return float(IMG_SIZE)
    return float(max(np.percentile(d1, 95), np.percentile(d2, 95)))


def calc_cbl(pred_bin: np.ndarray, gt_bin: np.ndarray):
    """CBL [0,1] cho mảng dự đoán. Trả về None nếu GT rỗng."""
    if gt_bin.sum() == 0:
        return None
    ys, xs = np.where(gt_bin)
    gt_diag = np.sqrt((ys.max() - ys.min()) ** 2 +
                      (xs.max() - xs.min()) ** 2) + 1e-6
    if pred_bin.sum() == 0:
        return 0.0
    yp, xp = np.where(pred_bin)
    d = np.sqrt((xp.mean() - xs.mean()) ** 2 +
                (yp.mean() - ys.mean()) ** 2)
    return float(np.clip(1.0 - d / gt_diag, 0.0, 1.0))

def calc_cbl_point(px: float, py: float, gt_bin: np.ndarray):
    """CBL [0,1] đo độ lệch từ 1 điểm (ngôi sao gợi ý) tới tâm GT."""
    if gt_bin.sum() == 0:
        return None
    ys, xs = np.where(gt_bin)
    gt_diag = np.sqrt((ys.max() - ys.min()) ** 2 +
                      (xs.max() - xs.min()) ** 2) + 1e-6
    cx_gt = float(xs.mean())
    cy_gt = float(ys.mean())
    d = np.sqrt((px - cx_gt) ** 2 + (py - cy_gt) ** 2)
    return float(np.clip(1.0 - d / gt_diag, 0.0, 1.0))


def get_centroid(binary_map: np.ndarray):
    """
    Tâm của binary mask trên 512x512.
    cx = trục ngang (cột), cy = trục dọc (hàng).
    Trả về (None, None) nếu mask rỗng.
    """
    if binary_map.sum() == 0:
        return None, None
    ys, xs = np.where(binary_map)
    return float(xs.mean()), float(ys.mean())


# =========================================================
# GRADCAM TỪ BOTTLENECK
# =========================================================

def compute_gradcam(model: torch.nn.Module,
                    image_tensor: torch.Tensor) -> np.ndarray | None:
    """
    GradCAM từ model.center (bottleneck) với zero_prompt.
    → Tìm vùng model 'thấy u' mà không bị prompt ảnh hưởng.

    Trả về numpy (H, W) ∈ [0,1] hoặc None nếu thất bại.
    """
    gradients:  list[torch.Tensor] = []
    activations: list[torch.Tensor] = []

    def fwd_hook(module, inp, out):
        activations.append(out)
        # Đăng ký hook trên tensor output để capture gradient
        out.register_hook(lambda g: gradients.append(g))

    hook = model.center.register_forward_hook(fwd_hook)
    model.eval()

    img_t       = image_tensor.clone().detach().to(DEVICE)
    zero_prompt = torch.zeros(1, 1, IMG_SIZE, IMG_SIZE, device=DEVICE)

    # Forward không có prompt → model tự tìm u
    out = model(img_t, zero_prompt)
    model.zero_grad()
    out.sum().backward()   # Gradient của toàn bộ output
    hook.remove()

    if not gradients or not activations:
        return None

    g = gradients[0]    # (1, C, h, w)
    a = activations[0]  # (1, C, h, w)
    # Trọng số theo gradient (Global Average Pooling)
    w   = g.mean(dim=(2, 3), keepdim=True)
    # Weighted sum + ReLU
    cam = F.relu((w * a).sum(dim=1, keepdim=True))
    # Upsample về 512x512
    cam = F.interpolate(cam, size=(IMG_SIZE, IMG_SIZE),
                        mode='bilinear', align_corners=False)
    cam = cam[0, 0].detach().cpu().numpy()
    # Normalize [0,1]
    vmin, vmax = cam.min(), cam.max()
    cam = (cam - vmin) / (vmax - vmin + 1e-8)
    return cam


# =========================================================
# PREDICT + KIỂM TRA PROMPT
# =========================================================

def predict_with_check(model:         torch.nn.Module,
                       img_tensor:    torch.Tensor,
                       prompt_tensor: torch.Tensor) -> dict:
    """
    Chạy inference và kiểm tra độ tin cậy.
    """
    model.eval()
    img_t = img_tensor.to(DEVICE)
    pmt_t = prompt_tensor.to(DEVICE)

    with torch.no_grad():
        out  = model(img_t, pmt_t)
        prob = torch.sigmoid(out)
        pred = (prob > 0.5).float()

    confidence = prob.max().item()
    pred_np    = pred[0, 0].cpu().numpy()
    pmt_np     = prompt_tensor[0, 0].cpu().numpy()

    # Tâm prompt (trên 512x512)
    ys_p, xs_p = np.where(pmt_np > 0.3)
    cx_pmt = float(xs_p.mean()) if len(xs_p) > 0 else IMG_SIZE / 2
    cy_pmt = float(ys_p.mean()) if len(ys_p) > 0 else IMG_SIZE / 2

    # Tâm dự đoán (trên 512x512)
    cx_pred, cy_pred = get_centroid(pred_np.astype(bool))

    # Khoảng cách tâm pred → tâm prompt
    if cx_pred is not None:
        center_dist = float(
            np.sqrt((cx_pred - cx_pmt) ** 2 + (cy_pred - cy_pmt) ** 2)
        )
    else:
        center_dist = float(IMG_SIZE)   # Mask rỗng → khoảng cách max

    # Kiểm tra nghi ngờ
    is_suspicious = (
        confidence < CONFIDENCE_THRESHOLD or
        center_dist > IMG_SIZE * CENTER_DIST_RATIO
    )

    # GradCAM chỉ khi nghi ngờ (tốn tính toán)
    saliency = compute_gradcam(model, img_tensor) if is_suspicious else None

    return {
        'mask':          pred_np,
        'prob_map':      prob[0, 0].cpu().numpy(),
        'confidence':    confidence,
        'center_dist':   center_dist,
        'cx_pred':       cx_pred,
        'cy_pred':       cy_pred,
        'is_suspicious': is_suspicious,
        'saliency':      saliency,
    }


# =========================================================
# VISUALIZE 1 ẢNH: PROMPT ĐÚNG vs PROMPT SAI
# =========================================================

def visualize_inference_check(model: torch.nn.Module, img_name: str):
    # ── Đọc file ──────────────────────────────────────────
    img_path  = os.path.join(TEST_IMAGE_DIR, img_name)
    json_path = os.path.join(TEST_JSON_DIR,
                             os.path.splitext(img_name)[0] + '.json')
    if not os.path.exists(img_path):
        print(f"  ⚠  Không tìm thấy: {img_path}")
        return

    image_cv = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)
    orig_h, orig_w = image_cv.shape

    with open(json_path, 'r', encoding='utf-8') as f:
        data = json.load(f)

    # Lấy polygon đầu tiên hợp lệ
    target_shape = next(
        (s for s in data.get('shapes', [])
         if s.get('shape_type') == 'polygon' and len(s.get('points', [])) > 2),
        data['shapes'][0] if data.get('shapes') else None
    )
    if target_shape is None:
        print(f"  ⚠  Không tìm thấy polygon trong {img_name}")
        return

    points = np.array(target_shape['points'])
    x_min, y_min = np.min(points, axis=0)
    x_max, y_max = np.max(points, axis=0)
    gt_w,  gt_h  = x_max - x_min, y_max - y_min

    mask_cv = np.zeros((orig_h, orig_w), dtype=np.uint8)
    cv2.fillPoly(mask_cv, [points.astype(np.int32)], 255)

    # Dataset helper để tạo heatmap đồng nhất
    ds_h = BTXRD_Dataset(TEST_IMAGE_DIR, TEST_JSON_DIR,
                         img_size=IMG_SIZE, is_train=False)
    r = (ds_h.zoom_ratio[0] + ds_h.zoom_ratio[1]) / 2

    def make_prompt(bx0, bx1, by0, by1):
        """Tạo heatmap + tensor. Trả về (numpy 512x512, tensor)."""
        pm  = ds_h.create_plateau_heatmap([bx0, by0, bx1, by1], orig_h, orig_w)
        pmr = cv2.resize(pm, (IMG_SIZE, IMG_SIZE))
        pmt = torch.from_numpy(pmr).unsqueeze(0).unsqueeze(0)  # (1,1,H,W)
        return pmr, pmt

    # Prompt ĐÚNG: zoom-out bao trọn GT
    pm_good_np, pm_good_t = make_prompt(
        max(0,      x_min - gt_w * r),
        min(orig_w, x_max + gt_w * r),
        max(0,      y_min - gt_h * r),
        min(orig_h, y_max + gt_h * r),
    )

    # Prompt SAI: đặt ở góc đối diện hoàn toàn với GT (không overlap)
    fx = orig_w - gt_w * (1 + r) - 10 if x_min < orig_w // 2 else 10
    fy = orig_h - gt_h * (1 + r) - 10 if y_min < orig_h // 2 else 10
    fx = max(0, fx)
    fy = max(0, fy)
    pm_bad_np, pm_bad_t = make_prompt(
        fx,
        min(orig_w, fx + gt_w * (1 + r)),
        fy,
        min(orig_h, fy + gt_h * (1 + r)),
    )

    # ── Resize ảnh & mask ─────────────────────────────────
    image_r = cv2.resize(image_cv, (IMG_SIZE, IMG_SIZE))
    mask_r  = cv2.resize(mask_cv,  (IMG_SIZE, IMG_SIZE),
                         interpolation=cv2.INTER_NEAREST)
    img_np  = image_r / 255.0
    gt_np   = (mask_r > 127).astype(np.float32)

    img_t = torch.from_numpy(
        (image_r.astype(np.float32) / 255.0 - 0.5) / 0.5
    ).unsqueeze(0).unsqueeze(0)

    # Tâm GT (512x512)
    cx_gt, cy_gt = get_centroid(gt_np.astype(bool))

    # ── Chạy predict ──────────────────────────────────────
    res_good = predict_with_check(model, img_t, pm_good_t)
    res_bad  = predict_with_check(model, img_t, pm_bad_t)

    # ── Vẽ figure ─────────────────────────────────────────
    fig, axes = plt.subplots(2, 5, figsize=(25, 10))
    enc_tag = 'ON' if USE_ENCODER_PROMPT else 'OFF'
    fig.suptitle(
        f"Cải tiến inference – Phát hiện prompt sai + GradCAM  |  {img_name}  |  encoder={enc_tag}\n"
        f"Ngưỡng: confidence < {CONFIDENCE_THRESHOLD}"
        f"  hoặc  dist tâm > {int(CENTER_DIST_RATIO*100)}% ảnh  →  kích hoạt GradCAM",
        fontsize=12, fontweight='bold'
    )

    col_titles = ["Ảnh gốc", "Prompt input", "Ground Truth",
                  "Dự đoán", "GradCAM / Prob map"]
    for c, t in enumerate(col_titles):
        axes[0][c].set_title(t, fontsize=11, fontweight='bold')

    rows_data = [
        ('Prompt ĐÚNG', res_good, pm_good_np, 'green'),
        ('Prompt SAI',  res_bad,  pm_bad_np,  'red'),
    ]

    for row, (label, res, pm_np, row_color) in enumerate(rows_data):
        status   = "✓ Tin cậy" if not res['is_suspicious'] else "⚠ Nghi ngờ → GradCAM"
        pred_m   = res['mask']
        cx_pred  = res['cx_pred']
        cy_pred  = res['cy_pred']

        # Tính metrics
        tp   = (pred_m * gt_np).sum()
        dice = (2 * tp + 1e-5) / (pred_m.sum() + gt_np.sum() + 1e-5)
        hd95 = calc_hd95(pred_m.astype(bool), gt_np.astype(bool))
        cbl  = calc_cbl(pred_m.astype(bool),  gt_np.astype(bool)) or 0.0

        # ── Col 0: Ảnh gốc ────────────────────────────────
        axes[row][0].imshow(img_np, cmap='gray')
        axes[row][0].set_ylabel(
            f"{label}\n{status}",
            fontsize=10, color=row_color, fontweight='bold',
            rotation=0, labelpad=82, va='center'
        )

        # ── Col 1: Prompt heatmap + info confidence ────────
        axes[row][1].imshow(img_np, cmap='gray')
        axes[row][1].imshow(
            np.ma.masked_where(pm_np < 0.05, pm_np),
            cmap='magma', alpha=0.65
        )
        axes[row][1].set_title(
            f"Conf: {res['confidence']:.3f}\nDist tâm: {res['center_dist']:.1f}px",
            fontsize=9
        )

        # ── Col 2: Ground Truth overlay + tâm GT ──────────
        axes[row][2].imshow(img_np, cmap='gray')
        green = np.zeros((*gt_np.shape, 4), dtype=np.float32)
        green[gt_np == 1] = [0, 1, 0, 0.35]
        axes[row][2].imshow(green)
        if gt_np.max() > 0:
            axes[row][2].contour(gt_np, [0.5], colors='lime', linewidths=1.5)
        if cx_gt is not None:
            axes[row][2].plot(
                cx_gt, cy_gt, 'o',
                color='lime', ms=9,
                markeredgecolor='black', markeredgewidth=1.2,
                label='Tâm GT'
            )
            axes[row][2].legend(fontsize=7, loc='lower right')

        # ── Col 3: Dự đoán overlay + tâm pred ─────────────
        axes[row][3].imshow(img_np, cmap='gray')
        red = np.zeros((*pred_m.shape, 4), dtype=np.float32)
        red[pred_m == 1] = [1, 0, 0, 0.35]
        axes[row][3].imshow(red)
        if pred_m.max() > 0:
            axes[row][3].contour(pred_m, [0.5], colors='red', linewidths=1.5)
        if cx_pred is not None:
            axes[row][3].plot(
                cx_pred, cy_pred, 'o',
                color='red', ms=9,
                markeredgecolor='white', markeredgewidth=1.2,
                label='Tâm Pred'
            )
            axes[row][3].legend(fontsize=7, loc='lower right')
        axes[row][3].set_title(
            f"Dice:{dice:.3f} | HD95:{hd95:.1f}px | CBL:{cbl:.3f}",
            fontsize=9, fontweight='bold'
        )

        # ── Col 4: GradCAM (nếu nghi ngờ) hoặc Prob map ───
        axes[row][4].imshow(img_np, cmap='gray')
        if res['saliency'] is not None:
            sal = res['saliency']
            axes[row][4].imshow(
                np.ma.masked_where(sal < 0.20, sal),
                cmap='jet', alpha=0.55
            )
            
            # Tính CBL cho gợi ý
            peak_y, peak_x = np.unravel_index(sal.argmax(), sal.shape)
            sugg_cbl = calc_cbl_point(peak_x, peak_y, gt_np)
            
            axes[row][4].plot(
                peak_x, peak_y, '*',
                color='yellow', ms=14,
                markeredgecolor='black', markeredgewidth=0.8,
                label='Đỉnh CAM\n(gợi ý)'
            )
            
            title_str = "GradCAM  ★ = gợi ý"
            if sugg_cbl is not None:
                title_str += f"\nCBL gợi ý: {sugg_cbl:.3f}"
                
            axes[row][4].set_title(
                title_str,
                fontsize=9, color='orange', fontweight='bold'
            )
            axes[row][4].legend(fontsize=7, loc='lower right')
        else:
            prob = res['prob_map']
            axes[row][4].imshow(
                np.ma.masked_where(prob < 0.10, prob),
                cmap='RdYlGn', alpha=0.60
            )
            axes[row][4].set_title("Probability map", fontsize=9)

        for ax in axes[row]:
            ax.axis('off')

    plt.tight_layout()
    plt.show()

# ENTRY POINT

if __name__ == "__main__":
    model = PGA_UNet(
        in_channels=1,
        n_classes=1,
        use_encoder_prompt=USE_ENCODER_PROMPT
    ).to(DEVICE)
    model.load_state_dict(
        torch.load(MODEL_PATH, map_location=DEVICE, weights_only=True)
    )
    model.eval()
    print(f"Model loaded: {MODEL_PATH}")
    print(f"Device: {DEVICE}  |  Encoder prompt: {'ON' if USE_ENCODER_PROMPT else 'OFF'}")
    print(f"Ngưỡng confidence: {CONFIDENCE_THRESHOLD}"
          f"  |  Ngưỡng dist tâm: {int(CENTER_DIST_RATIO*100)}% ảnh\n")

    for img_name in VIS_IMAGES:
        visualize_inference_check(model, img_name)
